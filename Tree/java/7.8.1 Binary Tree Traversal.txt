
package q73768;

import java.util.*;

class Node {
    int data;
    Node left, right;

    Node(int val) {
        data = val;
        left = null;
        right = null;
    }
}

public class CTJ73768 {

    static ArrayList<String> arr = new ArrayList<>();

    static Node buildTreeBFS(Node root, int ele) {
        Node temp = new Node(ele);

        if (root == null) {
            root = temp;
            return root;
        }

        Queue<Node> q = new LinkedList<>();
        q.add(root);

        while (!q.isEmpty()) {
            Node curr = q.poll();

            if (curr.left == null) {
                curr.left = temp;
                break;
            } else {
                q.add(curr.left);
            }

            if (curr.right == null) {
                curr.right = temp;
                break;
            } else {
                q.add(curr.right);
            }
        }

        return root;
    }

    static void inorder(Node root) {
        if (root == null)
            return;

        inorder(root.left);
        arr.add(root.data + " ");
        inorder(root.right);
    }

    static void preorder(Node root) {
        if (root == null)
            return;

        arr.add(root.data + " ");
        preorder(root.left);
        preorder(root.right);
    }

    static void postorder(Node root) {
        if (root == null)
            return;

        postorder(root.left);
        postorder(root.right);
        arr.add(root.data + " ");
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        Node root = null;

        for (int i = 0; i < n; i++) {
            int op = sc.nextInt();

            if (op == 1) {
                int val = sc.nextInt();
                root = buildTreeBFS(root, val);
            }

            else if (op == 2) {
                if (root == null)
                    arr.add("Empty\n");
                else {
                    inorder(root);
                    arr.add("\n");
                }
            }

            else if (op == 3) {
                if (root == null)
                    arr.add("Empty\n");
                else {
                    preorder(root);
                    arr.add("\n");
                }
            }

            else if (op == 4) {
                if (root == null)
                    arr.add("Empty\n");
                else {
                    postorder(root);
                    arr.add("\n");
                }
            }
        }

        for (String str : arr) {
            System.out.print(str);
        }

        sc.close();
    }
}
