
package q13912;

import java.util.*;

class Node {
    int data;
    Node left, right;

    Node(int val) {
        data = val;
        left = null;
        right = null;
    }
}

public class CTJ13912 {

    // Insert a node into BST
    static Node insert(Node root, int val) {
        if (root == null) {
            root = new Node(val);
            return root;
        }

        if (val < root.data)
            root.left = insert(root.left, val);

        else if (val > root.data)
            root.right = insert(root.right, val);

        return root;
    }

    // Check whether every non-leaf node has exactly one child
    static boolean checkPerfect(Node root) {
        if (root == null)
            return true;

        // If both children exist, condition is violated
        if (root.left != null && root.right != null)
            return false;

        return checkPerfect(root.left) &&
               checkPerfect(root.right);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        Node root = null;

        for (int i = 0; i < n; i++) {
            int x = sc.nextInt();
            root = insert(root, x);
        }

        System.out.println(checkPerfect(root) ? 1 : 0);

        sc.close();
    }
}
