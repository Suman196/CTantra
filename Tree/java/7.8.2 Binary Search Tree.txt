
package q17100;

import java.util.*;

class Node {
    int data;
    Node left, right;

    Node(int val) {
        data = val;
        left = null;
        right = null;
    }
}

public class CTJ17100 {

    static Node insert(Node root, int val) {
        if (root == null)
            return new Node(val);

        if (val < root.data)
            root.left = insert(root.left, val);
        else if (val > root.data)
            root.right = insert(root.right, val);

        return root;
    }

    static Node findMin(Node root) {
        while (root.left != null)
            root = root.left;

        return root;
    }

    static Node deleteNode(Node root, int val) {
        if (root == null)
            return null;

        if (val < root.data)
            root.left = deleteNode(root.left, val);

        else if (val > root.data)
            root.right = deleteNode(root.right, val);

        else {
            if (root.left == null)
                return root.right;

            if (root.right == null)
                return root.left;

            Node temp = findMin(root.right);
            root.data = temp.data;
            root.right = deleteNode(root.right, temp.data);
        }

        return root;
    }

    static boolean search(Node root, int val) {
        if (root == null)
            return false;

        if (root.data == val)
            return true;

        if (val < root.data)
            return search(root.left, val);

        return search(root.right, val);
    }

    static void inorder(Node root, ArrayList<Integer> ans) {
        if (root == null)
            return;

        inorder(root.left, ans);
        ans.add(root.data);
        inorder(root.right, ans);
    }

    static void preorder(Node root, ArrayList<Integer> ans) {
        if (root == null)
            return;

        ans.add(root.data);
        preorder(root.left, ans);
        preorder(root.right, ans);
    }

    static void postorder(Node root, ArrayList<Integer> ans) {
        if (root == null)
            return;

        postorder(root.left, ans);
        postorder(root.right, ans);
        ans.add(root.data);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        Node root = null;

        ArrayList<String> output = new ArrayList<>();

        while (n-- > 0) {
            int op = sc.nextInt();

            if (op == 1) {
                int val = sc.nextInt();
                root = insert(root, val);
            }

            else if (op == 2) {
                int val = sc.nextInt();
                root = deleteNode(root, val);
            }

            else if (op == 3) {
                int val = sc.nextInt();

                if (search(root, val))
                    output.add("found");
                else
                    output.add("not found");
            }

            else if (op >= 4 && op <= 6) {
                ArrayList<Integer> ans = new ArrayList<>();

                if (op == 4)
                    inorder(root, ans);

                else if (op == 5)
                    preorder(root, ans);

                else
                    postorder(root, ans);

                if (ans.isEmpty()) {
                    output.add("Empty");
                } else {
                    StringBuilder result = new StringBuilder();

                    for (int i = 0; i < ans.size(); i++) {
                        if (i > 0)
                            result.append(" ");

                        result.append(ans.get(i));
                    }

                    output.add(result.toString());
                }
            }
        }

        for (String str : output) {
            System.out.println(str);
        }

        sc.close();
    }
}
