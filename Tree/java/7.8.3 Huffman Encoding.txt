
package q31048;

import java.util.*;

class Node {
    int freq;
    char ch;
    Node left, right;

    Node(int frequency, char name) {
        freq = frequency;
        ch = name;
        left = null;
        right = null;
    }
}

public class CTJ31048 {

    static void preorder(Node root, StringBuilder temp, ArrayList<String> ans) {
        if (root == null)
            return;

        if (root.left == null && root.right == null) {
            ans.add(temp.toString());
            return;
        }

        temp.append('0');
        preorder(root.left, temp, ans);
        temp.deleteCharAt(temp.length() - 1);

        temp.append('1');
        preorder(root.right, temp, ans);
        temp.deleteCharAt(temp.length() - 1);
    }

    static ArrayList<String> huffmanCodes(String str, int[] f, int n) {

        PriorityQueue<Node> pq = new PriorityQueue<>(
            (a, b) -> Integer.compare(a.freq, b.freq)
        );

        for (int i = 0; i < n; i++) {
            pq.add(new Node(f[i], str.charAt(i)));
        }

        while (pq.size() > 1) {
            Node first = pq.poll();
            Node second = pq.poll();

            Node root = new Node(first.freq + second.freq, '$');

            root.left = first;
            root.right = second;

            pq.add(root);
        }

        Node root = pq.poll();

        ArrayList<String> ans = new ArrayList<>();
        StringBuilder temp = new StringBuilder();

        preorder(root, temp, ans);

        return ans;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String str = sc.next();

        int n = str.length();
        int[] f = new int[n];

        for (int i = 0; i < n; i++) {
            f[i] = sc.nextInt();
        }

        ArrayList<String> ans = huffmanCodes(str, f, n);

        for (String s : ans) {
            System.out.print(s + " ");
        }

        System.out.println();

        sc.close();
    }
}
